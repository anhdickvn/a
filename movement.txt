        let confirmPayload = MCVarInt.encode(teleport.0)
        movementDebug(String(format: "TELEPORT CONFIRM TX id=%d bytes=%d flags=0x%02X", teleport.0, confirmPayload.count, flagByte), force: true)
        send(packetId: 0x00, payload: confirmPayload)
    }

    private func dimensionName(_ id: Int32) -> String {
        switch id {
        case -1: return "Nether"
        case 1: return "The End"
        default: return "Overworld"
        }
    }

    // MARK: - WASD / Jump diagnostic

    private var movementDebugEnabled: Bool {
        UserDefaults.standard.bool(forKey: "mc_show_movement_debug")
    }

    private func movementDebug(_ text: String, force: Bool = false) {
        guard force || movementDebugEnabled else { return }
        let timestamp: String
        if let start = debugSessionStart {
            timestamp = String(format: "T+%.2f", Date().timeIntervalSince(start))
        } else {
            timestamp = String(format: "T+%.2f", Date().timeIntervalSince1970)
        }
        let line = "[\(timestamp)] \(text)"
        movementDebugBuffer.append(line)
        if movementDebugBuffer.count > 300 {
            movementDebugBuffer.removeFirst(movementDebugBuffer.count - 300)
        }
    }

    private func flushMovementDebug() {
        guard !movementDebugBuffer.isEmpty else { return }
        movementDebugLines = movementDebugBuffer
    }

    func clearMovementDebug() {
        movementDebugLines.removeAll(keepingCapacity: true)
        movementDebugBuffer.removeAll(keepingCapacity: true)
    }

    // MARK: - Điều khiển vị trí

    /// Một bước movement, dùng DUY NHẤT từ movement controller 20 Hz.
    /// Không cho nhiều timer/task cùng sửa tọa độ và gửi packet vì điều đó tạo
    /// burst movement và khiến server correction liên tục.
    func movePlayer(dx: Double, dz: Double) {
        guard state == .connected else {
            movementDebug("MOVE BLOCKED: state=\(state)")
            return
        }
        guard hasServerPosition else {
            movementDebug("MOVE BLOCKED: hasServerPosition=false")
            return
        }

        let now = Date().timeIntervalSince1970
        // Vanilla movement runs at 20 ticks/sec. Keep one authoritative sender.
        guard now - movementLastSentAt >= 0.049 else { return }
        let distance = hypot(dx, dz)
        guard distance > 0.0001 else { return }

        // Conservative walking speed: ~3.8 blocks/sec while holding a direction.
        // This avoids the old 4.7+ blocks/sec burst that could trip server correction.
        let maxStep = 0.19
        let scale = min(maxStep / distance, 1.0)
        let oldX = playerX
        let oldZ = playerZ
        playerX += dx * scale
        playerZ += dz * scale
        movementLastSentAt = now

        // Stay on the standard 1.12.2 Player Position packet for WASD.
        sendPlayerPosition(onGround: isOnGround)

        if movementDebugEnabled && now - lastMovementDebugAt >= 0.08 {
            lastMovementDebugAt = now
            movementDebug(String(format: "MOVE TX: (%.3f, %.3f) -> (%.3f, %.3f), yaw=%.1f, onGround=%@",
                                 oldX, oldZ, playerX, playerZ, playerYaw,
                                 isOnGround ? "true" : "false"))
        }
    }

    /// WASD theo yaw hiện tại của server.
    func movePlayer(forward: Double, strafe: Double) {
        guard state == .connected else {
            movementDebug("INPUT \(movementName(forward: forward, strafe: strafe)): state=\(state)")
            return
        }
        guard hasServerPosition else {
            movementDebug("INPUT \(movementName(forward: forward, strafe: strafe)): WAIT server position")
            return
        }

        let f = max(-1.0, min(1.0, forward))
        let s = max(-1.0, min(1.0, strafe))
        let length = hypot(f, s)
        guard length > 0 else { return }

        let yaw = Double(playerYaw).isFinite ? Double(playerYaw) : 0
        let yawRadians = yaw * .pi / 180.0
        let forwardX = -sin(yawRadians)
        let forwardZ = cos(yawRadians)
        let rightX = cos(yawRadians)
        let rightZ = sin(yawRadians)

        movementDebug("INPUT \(movementName(forward: forward, strafe: strafe)): yaw=\(String(format: "%.1f", yaw)) pos=(\(String(format: "%.2f", playerX)),\(String(format: "%.2f", playerY)),\(String(format: "%.2f", playerZ)))")
        movePlayer(dx: forwardX * (f / length) + rightX * (s / length),
                   dz: forwardZ * (f / length) + rightZ * (s / length))
    }

    private func movementName(forward: Double, strafe: Double) -> String {
        if forward > 0.5 && abs(strafe) < 0.5 { return "W" }
        if forward < -0.5 && abs(strafe) < 0.5 { return "S" }
        if strafe < -0.5 && abs(forward) < 0.5 { return "A" }
        if strafe > 0.5 && abs(forward) < 0.5 { return "D" }
        return "combo(\(forward),\(strafe))"
    }

    /// Một movement controller duy nhất. UIKit touch chỉ đổi trạng thái input;
    /// controller này là nơi DUY NHẤT tự động gửi movement packets.
    func startMoving(forward: Double, strafe: Double) {
        guard state == .connected else {
            movementDebug("TOUCH DOWN \(movementName(forward: forward, strafe: strafe)) nhưng state=\(state)", force: true)
            return
        }
        guard hasServerPosition else {
            movementDebug("TOUCH DOWN \(movementName(forward: forward, strafe: strafe)) nhưng chưa có server position", force: true)
            return
        }

        let key = "\(forward):\(strafe)"
        activeMovementInputs[key] = (forward, strafe)
        movementDebug("TOUCH DOWN \(movementName(forward: forward, strafe: strafe)); active=\(activeMovementInputs.count)", force: true)

        if movementControlTask == nil {
            movementControlTask = Task { @MainActor [weak self] in
                while !Task.isCancelled {
                    if self?.activeMovementInputs.isEmpty == false {
                        self?.sendCombinedMovementTick()
                    }
                    try? await Task.sleep(nanoseconds: 50_000_000)
                }
            }
        }
    }

    private func sendCombinedMovementTick() {
        guard state == .connected, hasServerPosition, !activeMovementInputs.isEmpty else { return }
        let forward = activeMovementInputs.values.reduce(0) { $0 + $1.forward }
        let strafe = activeMovementInputs.values.reduce(0) { $0 + $1.strafe }
        movePlayer(forward: max(-1, min(1, forward)), strafe: max(-1, min(1, strafe)))
    }

    func stopMoving(forward: Double, strafe: Double) {
        let key = "\(forward):\(strafe)"
        activeMovementInputs.removeValue(forKey: key)
        movementDebug("TOUCH UP \(movementName(forward: forward, strafe: strafe)); active=\(activeMovementInputs.count)", force: true)

        // Stop immediately. Do not run a post-release burst: a burst after release
        // was one of the sources of duplicate movement/correction loops.
        if activeMovementInputs.isEmpty {
            movementControlTask?.cancel()
            movementControlTask = nil
        }
    }

    /// Jump: one immediate impulse + 20 Hz gravity simulation.
    /// Never allow repeated jump tasks to overlap.
    func jumpPlayer() {
        guard state == .connected else {
            movementDebug("JUMP BLOCKED: state=\(state)", force: true)
            return
        }
        guard hasServerPosition else {
            movementDebug("JUMP BLOCKED: hasServerPosition=false", force: true)
            return
        }
        guard isOnGround, jumpTask == nil else {
            movementDebug("JUMP BLOCKED: onGround=\(isOnGround) jumpActive=\(jumpTask != nil)", force: true)
            return
        }

        let startY = playerY
        let startTime = Date().timeIntervalSince1970
        let generation = jumpGeneration &+ 1
        jumpGeneration = generation
        isOnGround = false
        movementDebug(String(format: "JUMP DOWN: startY=%.3f", startY), force: true)

        // Immediate jump impulse: the player moves right when SPACE is pressed.
        playerY = startY + 0.42
        movementLastSentAt = Date().timeIntervalSince1970
        sendPlayerPosition(onGround: false)

        let work = DispatchWorkItem { [weak self] in
            guard let self,
                  self.state == .connected,
                  self.jumpGeneration == generation else { return }

            let elapsed = Date().timeIntervalSince1970 - startTime
            let step = min(0.05, max(0.0, elapsed))
            let velocity = 4.3
            let gravity = 16.0
            let dy = velocity * step - 0.5 * gravity * step * step
            let newY = startY + 0.42 + dy

            if elapsed < 0.54 {
                self.playerY = newY
                self.movementLastSentAt = Date().timeIntervalSince1970
                self.sendPlayerPosition(onGround: false)
                self.jumpTask = DispatchWorkItem { [weak self] in
                    self?.jumpTask = nil
                    self?.jumpPlayerTick(generation: generation, startY: startY, startTime: startTime)
                }
                DispatchQueue.main.asyncAfter(deadline: .now() + 0.05, execute: self.jumpTask!)
            } else {
                self.playerY = startY
                self.isOnGround = true
                self.movementLastSentAt = Date().timeIntervalSince1970
                self.sendPlayerPosition(onGround: true)
                self.displayedPlayerX = self.playerX
                self.displayedPlayerY = self.playerY
                self.displayedPlayerZ = self.playerZ
                self.jumpTask = nil
            }
        }
        jumpTask = work
        DispatchQueue.main.async(execute: work)
    }

        let velocity = 4.3
        let gravity = 16.0

            playerY = startY
            isOnGround = true
